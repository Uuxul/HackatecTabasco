  <script>

    const video = document.getElementById("video");
    const canvas = document.getElementById("canvas");
    const ctx = canvas.getContext("2d", {
      willReadFrequently: true
    });

    const startButton = document.getElementById("startButton");

    const movementElement = document.getElementById("movement");
    const intensityElement = document.getElementById("intensity");
    const stateElement = document.getElementById("state");

    const statusText = document.getElementById("statusText");
    const dot = document.getElementById("dot");
    const warning = document.getElementById("warning");

    let previousFrame = null;
    let running = false;

    /*
     * Resolución utilizada para analizar la imagen.
     * No necesitamos analizar los 1920x1080 completos.
     */
    const WIDTH = 160;
    const HEIGHT = 120;

    startButton.addEventListener("click", startCamera);

    async function startCamera() {

      try {

        const stream = await navigator.mediaDevices.getUserMedia({
          video: {
            facingMode: "user"
          },
          audio: false
        });

        video.srcObject = stream;

        await video.play();

        running = true;

        statusText.textContent = "Cámara activa";
        dot.classList.remove("moving");

        startButton.textContent = "Cámara activa";
        startButton.disabled = true;

        canvas.width = WIDTH;
        canvas.height = HEIGHT;

        /*
         * Esperamos a que haya imagen antes de empezar.
         */
        setTimeout(() => {
          detectMovement();
        }, 500);

      } catch (error) {

        console.error(error);

        statusText.textContent = "Error al acceder a la cámara";

        alert(
          "No se pudo acceder a la cámara.\n\n" +
          "Comprueba que hayas permitido el acceso a la cámara " +
          "y que la página se esté ejecutando en HTTPS o localhost."
        );

      }
    }


    function detectMovement() {

      if (!running) return;

      /*
       * Dibujamos el frame actual en un canvas pequeño.
       */
      ctx.drawImage(
        video,
        0,
        0,
        WIDTH,
        HEIGHT
      );

      const currentFrame = ctx.getImageData(
        0,
        0,
        WIDTH,
        HEIGHT
      );

      /*
       * Si todavía no tenemos frame anterior,
       * guardamos el actual.
       */
      if (!previousFrame) {

        previousFrame = currentFrame;

        requestAnimationFrame(detectMovement);

        return;
      }


      let difference = 0;

      /*
       * Analizamos los píxeles.
       */
      for (
        let i = 0;
        i < currentFrame.data.length;
        i += 4
      ) {

        const r1 = currentFrame.data[i];
        const g1 = currentFrame.data[i + 1];
        const b1 = currentFrame.data[i + 2];

        const r2 = previousFrame.data[i];
        const g2 = previousFrame.data[i + 1];
        const b2 = previousFrame.data[i + 2];

        const diff =
          Math.abs(r1 - r2) +
          Math.abs(g1 - g2) +
          Math.abs(b1 - b2);

        /*
         * Ignoramos pequeñas variaciones
         * producidas por ruido de la cámara.
         */
        if (diff > 45) {
          difference++;
        }
      }


      const totalPixels =
        WIDTH * HEIGHT;

      /*
       * Porcentaje de píxeles que cambiaron.
       */
      let percentage =
        (difference / totalPixels) * 100;

      /*
       * Limitamos entre 0 y 100.
       */
      percentage =
        Math.min(100, percentage);

      /*
       * Suavizamos un poco el resultado.
       */
      percentage =
        Math.round(percentage);


      updateInterface(percentage);

      previousFrame = currentFrame;

      requestAnimationFrame(detectMovement);
    }


    function updateInterface(value) {

      movementElement.textContent =
        value + "%";


      /*
       * Clasificación sencilla del movimiento.
       *
       * IMPORTANTE:
       * Esto NO significa "agresividad".
       * Solamente mide cuánto cambió la imagen.
       */

      if (value < 3) {

        intensityElement.textContent =
          "Ninguna";

        intensityElement.className =
          "value normal";

        stateElement.textContent =
          "Normal";

        stateElement.className =
          "value normal";

        dot.classList.remove("moving");

        warning.style.display =
          "none";

      }

      else if (value < 12) {

        intensityElement.textContent =
          "Baja";

        intensityElement.className =
          "value normal";

        stateElement.textContent =
          "Actividad";

        stateElement.className =
          "value normal";

        dot.classList.add("moving");

        warning.style.display =
          "none";

      }

      else if (value < 25) {

        intensityElement.textContent =
          "Media";

        intensityElement.className =
          "value medium";

        stateElement.textContent =
          "Movimiento alto";

        stateElement.className =
          "value medium";

        dot.classList.add("moving");

        warning.style.display =
          "block";

      }

      else {

        intensityElement.textContent =
          "Alta";

        intensityElement.className =
          "value high";

        stateElement.textContent =
          "Movimiento intenso";

        stateElement.className =
          "value high";

        dot.classList.add("moving");

        warning.style.display =
          "block";
      }
    }
